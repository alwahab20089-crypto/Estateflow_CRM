import { ArrowLeft, History } from 'lucide-react';
import { useState } from 'react';
import { Link, useParams } from 'react-router-dom';
import { EmptyState } from '@/components/common/empty-state';
import { ErrorState } from '@/components/common/error-state';
import { PageHeader } from '@/components/common/page-header';
import { Pagination } from '@/components/common/pagination';
import { Button } from '@/components/ui/button';
import { Card } from '@/components/ui/card';
import { Skeleton } from '@/components/ui/skeleton';
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
import { errorMessage } from '@/lib/errors';
import { formatDateTime } from '@/lib/format';
import { TRIGGER_MAP } from './automation-catalog';
import { ExecutionStatusBadge, Pick, WorkflowStatusBadge } from './automation-ui';
import { ExecutionDetail } from './execution-detail';
import { useExecutions, useWorkflow } from './use-automations';

const STATUSES = ['PENDING', 'RUNNING', 'COMPLETED', 'PARTIALLY_FAILED', 'FAILED', 'CANCELLED'].map((s) => ({ value: s, label: s.replace('_', ' ').toLowerCase().replace(/^./, (c) => c.toUpperCase()) }));

export function ExecutionHistoryPage() {
  const { workflowId = '' } = useParams();
  const [page, setPage] = useState(1);
  const [status, setStatus] = useState('');
  const [selected, setSelected] = useState<string | null>(null);
  const workflow = useWorkflow(workflowId);
  const list = useExecutions(workflowId, { page, limit: 20, status: status || undefined });
  const items = list.data?.data ?? [];
  const meta = list.data?.meta;

  return (
    <>
      <PageHeader title={workflow.data ? `${workflow.data.name}: history` : 'Execution history'}
        description={workflow.data ? TRIGGER_MAP[workflow.data.triggerType].label : undefined}
        actions={<Button variant="outline" asChild><Link to="/app/automations"><ArrowLeft className="mr-1 size-4" /> All automations</Link></Button>} />
      {workflow.data && <div className="mb-3"><WorkflowStatusBadge status={workflow.data.status} /></div>}

      <div className="mb-4"><Pick aria-label="Filter by execution status" className="w-auto" placeholder="All statuses" options={STATUSES} value={status} onChange={(e) => { setStatus(e.target.value); setPage(1); }} /></div>

      <Card>
        {list.isLoading ? (
          <div className="space-y-2 p-4" role="status" aria-label="Loading executions">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-10 w-full" />)}</div>
        ) : list.isError ? (
          <ErrorState message={errorMessage(list.error)} onRetry={() => list.refetch()} />
        ) : items.length === 0 ? (
          <EmptyState icon={History} title={status ? 'No executions with this status' : 'No executions yet'} description={status ? undefined : 'When this automation runs, every run shows up here with its results.'} />
        ) : (
          <Table>
            <TableHeader><TableRow><TableHead>Started</TableHead><TableHead>Status</TableHead><TableHead>Attempts</TableHead><TableHead>Summary</TableHead></TableRow></TableHeader>
            <TableBody>
              {items.map((e) => (
                <TableRow key={e._id} className="cursor-pointer" onClick={() => setSelected(e._id)}>
                  <TableCell><button type="button" className="font-medium hover:underline" onClick={() => setSelected(e._id)}>{formatDateTime(e.startedAt ?? e.createdAt)}</button></TableCell>
                  <TableCell><ExecutionStatusBadge status={e.status} /></TableCell>
                  <TableCell>{e.attemptCount}/{e.maxAttempts}</TableCell>
                  <TableCell className="max-w-sm truncate text-sm text-muted-foreground">{e.errorSummary ?? `${e.actionResults.filter((r) => r.status === 'SUCCESS').length} of ${e.actionResults.length} actions succeeded`}</TableCell>
                </TableRow>
              ))}
            </TableBody>
          </Table>
        )}
      </Card>
      {meta && meta.total > 0 && <div className="mt-4"><Pagination page={meta.page} pageCount={meta.totalPages} total={meta.total} pageSize={meta.limit} onPageChange={setPage} /></div>}
      <ExecutionDetail workflowId={workflowId} executionId={selected} onClose={() => setSelected(null)} />
    </>
  );
}

import { RotateCcw } from 'lucide-react';
import { toast } from 'sonner';
import { ErrorState } from '@/components/common/error-state';
import { LoadingButton } from '@/components/common/loading-button';
import { Sheet, SheetContent, SheetDescription, SheetHeader, SheetTitle } from '@/components/ui/sheet';
import { Skeleton } from '@/components/ui/skeleton';
import { errorMessage } from '@/lib/errors';
import { formatDateTime } from '@/lib/format';
import { humanize } from '@/lib/labels';
import { useAuth } from '@/providers/auth-provider';
import type { ActionResult, ExecutionDto } from '@/types/automations';
import { ACTION_MAP, TRIGGER_MAP, describeAction } from './automation-catalog';
import { ExecutionStatusBadge, ResultIcon } from './automation-ui';
import { useExecution, useRetryExecution } from './use-automations';

/** Plain-language outcome, including notification delivery and the email provider's behaviour. */
export function resultNote(r: ActionResult): string {
  if (r.status === 'FAILED') return r.error ?? 'Failed';
  if (r.status === 'SKIPPED') return `Skipped: ${humanize(r.reason ?? 'not applicable')}`;
  if (r.status === 'PENDING') return 'Not run yet';
  const out = r.output ?? {};
  if (r.type === 'CREATE_IN_APP_NOTIFICATION') return out.created === false ? 'Not delivered (already delivered, or the recipient turned it off)' : 'Notification delivered';
  if (r.type === 'SEND_EMAIL_NOTIFICATION') return out.duplicate ? 'Already sent earlier: not sent again' : 'Email sent through the configured provider';
  return 'Done';
}

const FAILED = ['FAILED', 'PARTIALLY_FAILED'];

export function ExecutionDetail({ workflowId, executionId, onClose }: { workflowId: string; executionId: string | null; onClose: () => void }) {
  const { can } = useAuth();
  const q = useExecution(workflowId, executionId);
  const retry = useRetryExecution();
  const e: ExecutionDto | undefined = q.data;

  const doRetry = async () => {
    try { await retry.mutateAsync({ id: workflowId, executionId: executionId! }); toast.success('Retry queued. Only the failed actions will run again.'); }
    catch (err) { toast.error(errorMessage(err)); }
  };

  return (
    <Sheet open={Boolean(executionId)} onOpenChange={(o) => !o && onClose()}>
      <SheetContent className="w-full overflow-y-auto sm:max-w-lg">
        <SheetHeader>
          <SheetTitle>Execution details</SheetTitle>
          <SheetDescription>{e ? TRIGGER_MAP[e.triggerType].label : 'Loading…'}</SheetDescription>
        </SheetHeader>
        <div className="space-y-5 px-4 pb-6 text-sm">
          {q.isLoading && <Skeleton className="h-40 w-full" />}
          {q.isError && <ErrorState message={errorMessage(q.error)} onRetry={() => q.refetch()} />}
          {e && (
            <>
              <div className="flex items-center gap-2"><ExecutionStatusBadge status={e.status} /><span className="text-muted-foreground">Attempt {e.attemptCount} of {e.maxAttempts}{e.retryCount > 0 ? ` · retried manually ${e.retryCount}×` : ''}</span></div>
              {e.errorSummary && <p role="alert" className="rounded-md border border-destructive/30 bg-destructive/5 p-3 text-destructive">{e.errorSummary}</p>}
              {e.status === 'PENDING' && e.nextRetryAt && <p className="text-muted-foreground">Next attempt after {formatDateTime(e.nextRetryAt)}.</p>}
              <dl className="grid grid-cols-2 gap-2">
                <dt className="text-muted-foreground">Triggered by</dt><dd>{String(e.payload?.name ?? '—')}</dd>
                <dt className="text-muted-foreground">Created</dt><dd>{formatDateTime(e.createdAt)}</dd>
                {e.completedAt && <><dt className="text-muted-foreground">Finished</dt><dd>{formatDateTime(e.completedAt)}</dd></>}
                <dt className="text-muted-foreground">Version</dt><dd>{e.workflowVersion}</dd>
              </dl>
              <section aria-label="Action results">
                <h3 className="mb-2 font-semibold">Actions</h3>
                <ol className="space-y-2">
                  {e.actionResults.map((r) => (
                    <li key={r.index} className="flex gap-2 rounded-md border p-3">
                      <ResultIcon status={r.status} />
                      <div className="min-w-0">
                        <p className="font-medium">{ACTION_MAP[r.type].label} <span className="font-normal text-muted-foreground">· {humanize(r.status)}{r.attempts > 1 ? ` · ${r.attempts} attempts` : ''}</span></p>
                        {e.actions?.[r.index] && <p className="text-muted-foreground">{describeAction(e.actions[r.index]!)}</p>}
                        <p className={r.status === 'FAILED' ? 'text-destructive' : 'text-muted-foreground'}>{resultNote(r)}</p>
                      </div>
                    </li>
                  ))}
                </ol>
              </section>
              {FAILED.includes(e.status) && can('automations:retry') && (
                <LoadingButton loading={retry.isPending} onClick={doRetry}><RotateCcw className="mr-1 size-4" /> Retry failed actions</LoadingButton>
              )}
            </>
          )}
        </div>
      </SheetContent>
    </Sheet>
  );
}

import { screen, waitFor, within } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { beforeEach, describe, expect, it, vi } from 'vitest';
import { automationService } from '@/services/automation-service';
import { renderWithProviders } from '../../test/utils';
import type { ExecutionDto } from '@/types/automations';
import { ExecutionHistoryPage } from './execution-history-page';

const { perms } = vi.hoisted(() => ({
    perms: vi.fn<(permission: string) => boolean>(() => true),
}));

vi.mock('@/services/automation-service', () => ({
    automationService: {
        get: vi.fn(),
        executions: vi.fn(),
        execution: vi.fn(),
        retry: vi.fn(),
    },
    notificationPreferenceService: {},
}));

vi.mock('sonner', () => ({
    toast: {
        success: vi.fn(),
        error: vi.fn(),
    },
}));

vi.mock('@/providers/auth-provider', () => ({
    useAuth: () => ({
        can: (permission: string) => perms(permission),
    }),
}));

const exec = (o: Partial<ExecutionDto> = {}): ExecutionDto => ({
    _id: 'e1',
    workflowId: 'w1',
    workflowVersion: 2,
    triggerEventId: 'ev1',
    triggerType: 'LEAD_CREATED',
    status: 'PARTIALLY_FAILED',
    attemptCount: 3,
    maxAttempts: 3,
    retryCount: 0,
    startedAt: '2026-11-16T10:00:00.000Z',
    completedAt: '2026-11-16T10:05:00.000Z',
    nextRetryAt: null,
    errorSummary: 'SEND_EMAIL_NOTIFICATION: smtp down',
    createdAt: '2026-11-16T10:00:00.000Z',
    actions: [
        {
            type: 'CREATE_IN_APP_NOTIFICATION',
            recipient: { mode: 'ASSIGNEE' },
            title: 'Hi',
            message: 'm',
        },
        {
            type: 'SEND_EMAIL_NOTIFICATION',
            recipient: { mode: 'ASSIGNEE' },
            subject: 'Lead',
            body: 'b',
        },
    ],
    payload: { name: 'Olivia Bennett' },
    actionResults: [
        {
            index: 0,
            type: 'CREATE_IN_APP_NOTIFICATION',
            status: 'SUCCESS',
            attempts: 1,
            output: { created: true },
        },
        {
            index: 1,
            type: 'SEND_EMAIL_NOTIFICATION',
            status: 'FAILED',
            attempts: 3,
            error: 'smtp down',
            retryable: true,
        },
    ],
    ...o,
});

const page = (data: ExecutionDto[]) => ({
    data,
    meta: {
        page: 1,
        limit: 20,
        total: data.length,
        totalPages: 1,
    },
});

const render = () =>
    renderWithProviders(<ExecutionHistoryPage />, {
        route: '/app/automations/w1/executions',
        path: '/app/automations/:workflowId/executions',
    });

beforeEach(() => {
    vi.clearAllMocks();
    perms.mockReturnValue(true);

    vi.mocked(automationService.get).mockResolvedValue({
        _id: 'w1',
        name: 'Follow up',
        status: 'ACTIVE',
        triggerType: 'LEAD_CREATED',
    } as never);
});

describe('ExecutionHistoryPage', () => {
    it('shows a failed execution with per-action results and delivery behaviour', async () => {
        vi.mocked(automationService.executions).mockResolvedValue(page([exec()]));
        vi.mocked(automationService.execution).mockResolvedValue(exec());

        render();

        await userEvent.click(await screen.findByText('Partially failed'));

        const results = await screen.findByRole('region', {
            name: 'Action results',
        });

        expect(
            within(results).getByText('Notification delivered'),
        ).toBeInTheDocument();

        expect(within(results).getAllByText('smtp down').length).toBeGreaterThan(0);
        expect(within(results).getByText(/3 attempts/)).toBeInTheDocument();
        expect(screen.getByText('Olivia Bennett')).toBeInTheDocument();
    });

    it('retries a failed execution once and refreshes', async () => {
        vi.mocked(automationService.executions).mockResolvedValue(page([exec()]));
        vi.mocked(automationService.execution).mockResolvedValue(exec());

        vi.mocked(automationService.retry).mockResolvedValue(
            exec({ status: 'PENDING' }),
        );

        render();

        await userEvent.click(await screen.findByText('Partially failed'));

        await userEvent.click(
            await screen.findByRole('button', {
                name: /Retry failed actions/,
            }),
        );

        await waitFor(() =>
            expect(automationService.retry).toHaveBeenCalledWith('w1', 'e1'),
        );
    });

    it('offers no retry for completed executions or without permission', async () => {
        const completedExecution = exec({
            status: 'COMPLETED',
            errorSummary: null,
            actionResults: [
                {
                    index: 0,
                    type: 'CREATE_IN_APP_NOTIFICATION',
                    status: 'SUCCESS',
                    attempts: 1,
                    output: { created: false },
                },
            ],
        });

        vi.mocked(automationService.executions).mockResolvedValue(
            page([completedExecution]),
        );

        vi.mocked(automationService.execution).mockResolvedValue(
            completedExecution,
        );

        render();

        await userEvent.click(await screen.findByText('Completed'));

        expect(await screen.findByText(/Not delivered/)).toBeInTheDocument();

        // Suppressed by preference or duplicate.
        expect(
            screen.queryByRole('button', { name: /Retry/ }),
        ).not.toBeInTheDocument();
    });

    it('hides retry without the retry permission', async () => {
        perms.mockImplementation(
            (permission) => permission !== 'automations:retry',
        );

        vi.mocked(automationService.executions).mockResolvedValue(page([exec()]));
        vi.mocked(automationService.execution).mockResolvedValue(exec());

        render();

        await userEvent.click(await screen.findByText('Partially failed'));

        await screen.findByRole('region', {
            name: 'Action results',
        });

        expect(
            screen.queryByRole('button', { name: /Retry/ }),
        ).not.toBeInTheDocument();
    });

    it('renders empty and error states', async () => {
        vi.mocked(automationService.executions).mockResolvedValueOnce(page([]));

        const { unmount } = render();

        expect(await screen.findByText('No executions yet')).toBeInTheDocument();

        unmount();

        vi.mocked(automationService.executions).mockRejectedValueOnce(
            new Error('boom'),
        );

        render();

        expect(await screen.findByRole('alert')).toBeInTheDocument();
    });
});
